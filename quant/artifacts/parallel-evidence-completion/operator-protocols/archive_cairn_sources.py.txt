"""Archive primary leads for the newly observed held-security blocker; no admission."""
from pathlib import Path
from datetime import datetime, timezone
import hashlib
import json
import requests
from quant.storage.research_store import write_json, digest_file

root = Path('data/parallel-evidence-completion').resolve()
sources = [
 ('merger-effective', 'https://www.vedantalimited.com/public/uploads/11865/vedanta-cairn-merger-effective-april-11-2017.pdf'),
 ('allotment', 'https://www.vedantalimited.com/uploads/stock-exchange-announcements/2017-18/Allotment-of-shares-of-Vedanta-to-shareholders-of-Cairn-.India-Limited.pdf'),
 ('quarter-update', 'https://www.vedantalimited.com/public/uploads/5033/Vedanta-Limited-Statement-Of-Unaudited-Standalone-Results-For-The-Quarter-Ended-June-30-2017.pdf'),
 ('sec-merger-exhibit','https://www.sec.gov/Archives/edgar/data/1370431/000119312517120784/d270284dex991.htm')]
records = []
for name,url in sources:
    item = dict(source_url=url, retrieved_at=datetime.now(timezone.utc).isoformat(),
        retrieved_at_used_as_historical_availability=False)
    try:
        response = requests.get(url,timeout=20,headers={'User-Agent':'MarketPulse research evidence archiver'})
        body = response.content
        digest = hashlib.sha256(body).hexdigest()
        suffix = '.pdf' if body.startswith(b'%PDF') else '.html'
        path = root/'raw/security-transitions/cairn-2017'/f'{name}-{digest[:16]}{suffix}'
        path.parent.mkdir(parents=True,exist_ok=True)
        if not path.exists():
            path.write_bytes(body)
        elif digest_file(path) != digest:
            raise ValueError('Archived source bytes changed')
        item.update(http_status=response.status_code,final_url=response.url,sha256=digest,
            raw_path=path.relative_to(root).as_posix(),content_type=response.headers.get('Content-Type'),
            usable_source_bytes=response.status_code==200 and (body.startswith(b'%PDF') or 'sec.gov' in url))
    except requests.RequestException as error:
        item.update(error=str(error),usable_source_bytes=False)
    records.append(item)
packet = dict(status='NEEDS_EVIDENCE_AND_ACCOUNTING_REVIEW',subject_id='CAIRN_2017',
    instrument_id='a4d2246c-e16f-54c3-a1d0-14c307f6ba60',observed_block_date='2017-04-26',
    observed_error='UnvaluedHolding',research_authority='DIAGNOSTIC_ONLY',seal_eligible=False,
    source_market_corpus_modified=False,TIDEWATER_human_review_status='PENDING',
    source_records=records,
    candidate_primary_source_facts=[
        dict(source='merger-effective',facts=['Issuer states merger became operative on 2017-04-11 and sets 2017-04-27 record date.',
            'Issuer describes one Vedanta equity share plus four redeemable preference shares per Cairn equity share.',
            'Issuer describes a separate interim dividend and an optional third-party preference-share cash exit facility.']),
        dict(source='allotment',facts=['Issuer reports allotment on 2017-04-28; its listing applications were still to be made.']),
        dict(source='quarter-update',facts=['Later issuer update states new equity shares traded from 2017-05-16.',
            'The later update describes preference shares credited with their listing application still pending.'])],
    facts_not_yet_certified=['Contemporaneous suspension and equity listing notices with historical availability.',
        'Exact credit dates and instrument/ISIN lineage for both consideration legs.',
        'Then-known preference-share valuation, any redemption/exit participation and cash/dividend timing.',
        'Accounting and cost-basis treatment across equity, preference and cash legs.'],
    restrictions=['No automatic one-for-one equity-only bridge.',
        'Optional cash exit is not an assumed executed settlement.',
        'Later disclosure dates do not backdate historical knowledge.',
        'Prepared fallback declaration is unchanged; no accepted transition or research seal is created.'])
write_json(root/'reports/security-transitions/cairn-2017-evidence-gap.json',packet)
print(json.dumps([dict(url=r['source_url'],status=r.get('http_status'),usable=r['usable_source_bytes']) for r in records],indent=2))
