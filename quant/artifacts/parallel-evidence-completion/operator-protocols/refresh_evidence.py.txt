"""Bounded operator run; retained data are read only and no human approvals are created."""
import json
from pathlib import Path
from dataclasses import replace
import pandas as pd
from quant.config import Config
from quant.storage.parquet_store import read_parquet_file, write_parquet
from quant.storage.research_store import write_json, digest_file
from quant.pipeline.audit_fundamental_coverage import audit_fundamental_coverage
from quant.pipeline.report_fundamental_coverage_ceiling import coverage_ceiling
from quant.data.fundamental_discovery_gap import build_discovery_gap
from quant.pipeline.discover_fundamentals_targeted import discover_targeted
from quant.pipeline.verify_capital_events import verify_capital_events
from quant.pipeline.audit_share_history_coverage import audit_share_history_coverage
from quant.pipeline.plan_share_evidence import plan_share_evidence
from quant.pipeline.report_historical_forward_phase import report_phase

root = Path('data/parallel-evidence-completion').resolve()
corpus = Path('D:/Market_Pulse_Quant_20261113/data')
config = replace(Config(), data_dir=root, timeout=12, retries=0)
filings = read_parquet_file(Path('data/evidence-closure/inputs/filings.parquet'))
universe = read_parquet_file(Path('data/evidence-closure/inputs/universe.parquet'))
attempts = json.loads(Path('data/evidence-closure/inputs/attempts.json').read_text())['attempted']
calendar = read_parquet_file(corpus/'normalized/calendar/nse_sessions.parquet')
sessions = calendar.loc[calendar.is_trading_session, 'trade_date']
history = read_parquet_file(corpus/'normalized/security_history/identities.parquet')
start, end = pd.Timestamp(universe.as_of.min()).date(), pd.Timestamp(universe.as_of.max()).date()
# Continue the checksum-retained prior input corpus; discover_targeted itself
# recomputes and verifies the supplied ceiling, then audits/replans the output.
coverage = json.loads(Path('data/evidence-closure/reports/fundamental-universe-coverage.json').read_text())
ceiling = json.loads(Path('data/historical-certification/reports/fundamental-coverage-ceiling.json').read_text())
write_json(root/'reports/pre-batch-coverage.json', coverage)
write_json(root/'reports/pre-batch-ceiling.json', ceiling)
# Reuse archived official metadata proofs. They do not constitute human governance approval.
old = Path('data/historical-forward-validation')
sources = json.loads((old/'fallback-manifest.json').read_text())
import shutil
for source in sources:
    name = source['dissemination_evidence_path']
    path = root/name
    path.parent.mkdir(parents=True, exist_ok=True)
    shutil.copyfile(old/name, path)
batch = discover_targeted(config, universe, coverage, ceiling, filings, sessions, history,
    attempts=attempts, fallback_manifest=sources, max_requests=5, max_documents=4)
write_json(root/'reports/latest-discovery-batch.json', batch)
output = Path(batch['output'])
coverage = json.loads((output/'coverage.json').read_text())
ceiling = json.loads((output/'coverage-ceiling.json').read_text())
combined = read_parquet_file(output/'filings.parquet')
write_json(root/'reports/fundamental-universe-coverage.json', coverage)
write_json(root/'reports/fundamental-coverage-ceiling.json', ceiling)
write_json(root/'reports/fundamental-discovery-gap.json', build_discovery_gap(universe, coverage, ceiling, combined))
anchors = read_parquet_file(Path('data/coverage-validation/inputs/anchors.parquet'))
events_path = Path('data/historical-certification/reports/capital-event-candidates-final.parquet')
events = verify_capital_events(read_parquet_file(events_path), corpus)
write_parquet(events, root/'reports/capital-events-reviewed.parquet')
write_json(root/'reports/capital-event-verification.json', dict(rows=len(events),
    status_counts=events.verification_status.value_counts().to_dict(), input_sha256=digest_file(events_path),
    output_sha256=digest_file(root/'reports/capital-events-reviewed.parquet'), complete_chains_certified=False))
intervals = read_parquet_file(Path('data/coverage-validation/inputs/share-coverage.parquet'))
shares = audit_share_history_coverage(universe, anchors, events, intervals, sessions, start=start, end=end)
write_json(root/'reports/share-history-coverage.json', shares)
write_json(root/'reports/share-evidence-plan.json', plan_share_evidence(universe, anchors, events, intervals,
    sessions, start=start, end=end, filing_coverage=coverage))
phase = report_phase(config, **{name: root/'reports'/file for name, file in dict(
    filing_coverage='fundamental-universe-coverage.json', share_coverage='share-history-coverage.json',
    discovery_gap='fundamental-discovery-gap.json', ceiling='fundamental-coverage-ceiling.json',
    capital_verification='capital-event-verification.json').items()},
    market_session_evidence=Path('quant/artifacts/historical-forward-validation/reports/historical-forward-validation-phase.json'))
print(json.dumps(dict(discovery=batch['status'], requests=batch['requests'], productivity=batch['productivity'],
    governance=phase['governance']['human_review_status'], shares=shares['status']), indent=2))
