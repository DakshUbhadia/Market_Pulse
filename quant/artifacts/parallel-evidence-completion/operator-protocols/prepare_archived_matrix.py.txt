import json
from pathlib import Path
from datetime import date
from quant.storage.parquet_store import read_parquet_file, write_parquet
from quant.storage.research_store import digest_file, write_json
from quant.storage.duckdb_store import query_dataset
from quant.pipeline.build_fundamental_features import build_fundamental_features
from quant.pipeline.prepare_v2_matrix import prepare_v2_matrix
from quant.pipeline.audit_fundamental_coverage import verify_filing_archives
from quant.pipeline.audit_share_history_coverage import verify_share_archives

root = Path('data/parallel-evidence-completion')
corpus = Path('D:/Market_Pulse_Quant_20261113/data')
feature_path = corpus/'features/factor_v1/year=2026/2026-11-17.parquet'
manifest = json.loads(feature_path.with_suffix('.manifest.json').read_text())
assert digest_file(feature_path) == manifest['feature_sha256']
features = read_parquet_file(feature_path)
for item in manifest['files']:
    path = (corpus/item['path']).resolve()
    assert path.is_relative_to(corpus.resolve()) and digest_file(path) == item['sha256']
filing_path = Path('data/evidence-closure/inputs/filings.parquet')
filings = verify_filing_archives(read_parquet_file(filing_path), corpus)
paths = [Path('data/coverage-validation/inputs')/name for name in ('anchors.parquet', 'capital-events.parquet', 'share-coverage.parquet')]
frames = [verify_share_archives(read_parquet_file(p), corpus) for p in paths]
prices = query_dataset(corpus, 'adjusted_prices', start=date(2026,11,17), end=date(2026,11,17))
keys = features[['as_of', 'instrument_id']].merge(prices.loc[prices.series.eq('EQ'), ['trade_date', 'instrument_id', 'raw_close']].rename(columns={'trade_date':'as_of'}), on=['as_of','instrument_id'], validate='one_to_one')
fundamentals = build_fundamental_features(keys, filings, *frames)
write_parquet(fundamentals, root/'inputs/archived-fundamental-features.parquet')
arms, report = prepare_v2_matrix(features, fundamentals)
report.update(input_scope='ARCHIVED_DIAGNOSTIC_INPUTS', archived_price_feature_implementation_sha=manifest['git_commit'],
    archived_price_features_sha256=digest_file(feature_path), source_manifest_sha256=digest_file(feature_path.with_suffix('.manifest.json')),
    current_implementation_regenerated_price_features=False, certified_interval=False,
    source_market_corpus_read_only=True, input_sha256={str(p):digest_file(p) for p in [filing_path,*paths]})
for name, frame in arms.items():
    write_parquet(frame, root/'matrix'/f'{name}.parquet')
write_json(root/'reports/v2-matrix-status.json', report)
print(json.dumps(report, indent=2))
